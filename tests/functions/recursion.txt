def fact(n)
    if n <= 1
        return 1
    end
    return n * fact(n - 1)
end

cout fact(0)
cout fact(1)
cout fact(5)
cout fact(6)

def fib(n)
    if n <= 0
        return 0
    elif n == 1
        return 1
    end
    return fib(n - 1) + fib(n - 2)
end

cout fib(0)
cout fib(1)
cout fib(6)
cout fib(7)
