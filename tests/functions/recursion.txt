def fact(n)
    if n <= 1
        return 1
    end
    return n * fact(n - 1)
end

Cout(fact(0))
Cout(fact(1))
Cout(fact(5))
Cout(fact(6))

def fib(n)
    if n <= 0
        return 0
    elif n == 1
        return 1
    end
    return fib(n - 1) + fib(n - 2)
end

Cout(fib(0))
Cout(fib(1))
Cout(fib(6))
Cout(fib(7))
