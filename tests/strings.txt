# String concatenation tests
set s1 = "Hello"
set s2 = "World"
set space = " "
set combined = s1 + space + s2
cout combined

set msg = "Number: " + 42
cout msg

set bool_msg = "Status: " + true
cout bool_msg

cout "Direct: " + (10 + 20)

# Equality and inequality tests
cout "apple" == "apple"
cout "apple" == "orange"
cout "apple" != "orange"
cout "42" == 42
cout "42" != 42

# Len function tests
cout Len("Antigravity")
cout Len("")
set txt = "12345"
cout Len(txt)
cout Len(txt) + 5

# ToUpper and ToLower tests
cout ToUpper("lowercase text")
cout ToLower("UPPERCASE TEXT")
set mixed = "HeLLo WoRLd"
cout ToUpper(mixed)
cout ToLower(mixed)

# Substr function tests
