# String concatenation tests
set s1 = "Hello"
set s2 = "World"
set space = " "
set combined = s1 + space + s2
Cout(combined)

set msg = "Number: " + 42
Cout(msg)

set bool_msg = "Status: " + true
Cout(bool_msg)

Cout("Direct: " + (10 + 20))

# Equality and inequality tests
Cout("apple" == "apple")
Cout("apple" == "orange")
Cout("apple" != "orange")
Cout("42" == 42)
Cout("42" != 42)

# Len function tests
Cout(Len("Antigravity"))
Cout(Len(""))
set txt = "12345"
Cout(Len(txt))
Cout(Len(txt) + 5)

# ToUpper and ToLower tests
Cout(ToUpper("lowercase text"))
Cout(ToLower("UPPERCASE TEXT"))
set mixed = "HeLLo WoRLd"
Cout(ToUpper(mixed))
Cout(ToLower(mixed))

# Substr function tests
set source = "Antigravity"
Cout(Substr(source, 0, 4))
Cout(Substr(source, 4, 7))
Cout(Substr(source, 4, 50))
Cout(Substr("Test", 0, 0))
