# Escape sequences: \n, \t, \", \', \\
set s1 = "Hello\nWorld"
Cout(s1)

set s2 = "Col1\tCol2"
Cout(s2)

set s3 = "He said \"hello\""
Cout(s3)

set s4 = 'She said \'hi\''
Cout(s4)

set s5 = "Backslash: \\"
Cout(s5)

# Array with escaped strings
set arr = ["line1\nline2", "tab\tbed", "\"quoted\""]
Cout(arr[0])
Cout(arr[1])
Cout(arr[2])

# Direct function call argument with escape
Cout("Direct: \"test\"\nNext line")
