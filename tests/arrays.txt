# Array initialization
set empty = []
Cout(empty)
Cout(Len(empty))
Cout(TypeOf(empty))

set numbers = [10, 20, 30]
Cout(numbers)
Cout(Len(numbers))
Cout(TypeOf(numbers))

# Mixed array
set mixed = [1, "hello", true]
Cout(mixed)

# Array indexing
Cout(numbers[0])
Cout(numbers[1])
Cout(numbers[2])
Cout(mixed[1])
Cout(mixed[2])

# Math operations with array elements
set sum = numbers[0] + numbers[1]
Cout(sum)
Cout(numbers[2] * 2)

# Index assignment
numbers[1] = 99
Cout(numbers)
Cout(numbers[1])

# Dynamic index via variable
set idx = 2
numbers[idx] = 100
Cout(numbers[2])
Cout(numbers)

# Array iteration
set items = [100, 200, 300]
for i = 0 to Len(items) - 1
    Cout(items[i])
end

# Push and Pop operations
set stack = []
Push(stack, 1)
Push(stack, 2)
Push(stack, 3)
Cout(stack)
Cout(Len(stack))

set top = Pop(stack)
Cout(top)
Cout(stack)
Cout(Len(stack))

# Array in function
def doubleList(list)
    set result = []
    for i = 0 to Len(list) - 1
        Push(result, list[i] * 2)
    end
    return result
end

set inputList = [1, 2, 3]
set doubledList = doubleList(inputList)
Cout(doubledList)
